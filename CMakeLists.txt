cmake_minimum_required(VERSION 3.24)
project(ArchaeologyWorkbench VERSION 0.5.4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
# CMake 4.3 puede interpretar el prefijo UTF-8 de MSVC en español como si usara la página OEM.
# Sin esta corrección, Ninja no registra cabeceras y puede enlazar objetos incompatibles.
if(MSVC AND CMAKE_CXX_CL_SHOWINCLUDES_PREFIX MATCHES "^Nota: inclusi")
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "Nota: inclusión del archivo: ")
endif()
include(CTest)
option(WORKBENCH_BUILD_APP "Build the desktop application and geodetic worker" ON)
option(WORKBENCH_BUILD_TRANSCRIPTION_WORKER "Build the native inference worker (requires its SDK)" ON)
option(WORKBENCH_BUILD_INTEGRATION_TESTS "Build tests requiring staged module engines" ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Sql Concurrent)
if(WORKBENCH_BUILD_APP)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 Network LinguistTools)
endif()
qt_standard_project_setup(REQUIRES 6.8)
add_library(WorkbenchCoordinateDomain STATIC modules/coordinates/CoordinateText.cpp)
target_link_libraries(WorkbenchCoordinateDomain PUBLIC Qt6::Core)
target_include_directories(WorkbenchCoordinateDomain PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(WorkbenchCore STATIC core/Manifest.cpp core/Paths.cpp core/ProjectTemplate.cpp)
target_include_directories(WorkbenchCore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(WorkbenchCore PUBLIC Qt6::Core)

add_library(WorkbenchServices STATIC
    services/Store.cpp services/Settings.cpp services/Logger.cpp
    services/Volumes.cpp services/TaskManager.cpp services/Registry.cpp
    modules/process/ProcessModule.cpp)
target_link_libraries(WorkbenchServices PUBLIC WorkbenchCore Qt6::Gui Qt6::Sql Qt6::Concurrent)

if(WORKBENCH_BUILD_APP)
set(WORKBENCH_PROJ_INCLUDE_DIR "" CACHE PATH "Directory containing the PROJ 9.x development header")
find_path(PROJ_HEADER_DIR proj.h HINTS ${WORKBENCH_PROJ_INCLUDE_DIR})
if(NOT PROJ_HEADER_DIR)
    message(FATAL_ERROR "PROJ development header required for the isolated coordinate worker; set WORKBENCH_PROJ_INCLUDE_DIR")
endif()
add_executable(CoordinateWorker modules/coordinates/worker.cpp)
target_include_directories(CoordinateWorker PRIVATE ${PROJ_HEADER_DIR})
target_link_libraries(CoordinateWorker PRIVATE WorkbenchCoordinateDomain Qt6::Core)
if(MSVC)
    target_compile_options(WorkbenchCoordinateDomain PRIVATE /utf-8 /W4)
    target_compile_options(CoordinateWorker PRIVATE /utf-8 /W4)
endif()

qt_add_executable(ArchaeoHub app/main.cpp app/UiChaos.cpp ui/Hub.cpp ui/ToolFilter.h ui/SymbolProvider.cpp
    modules/catalog/CatalogController.cpp modules/coordinates/CoordinateController.cpp
    modules/transcription/TranscriptionController.cpp modules/documents/DocumentController.cpp modules/documents/DocumentLibraryController.cpp modules/inventory/InventoryController.cpp resources.qrc)
if(WIN32)
    enable_language(RC)
    configure_file(resources/windows.rc.in ${CMAKE_CURRENT_BINARY_DIR}/workbench.rc @ONLY)
    target_sources(ArchaeoHub PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/workbench.rc)
endif()

if(WORKBENCH_BUILD_TRANSCRIPTION_WORKER)
set(WORKBENCH_TRANSCRIPTION_SDK "" CACHE PATH "Existing native transcription SDK and compiled inference libraries")
if(NOT WORKBENCH_TRANSCRIPTION_SDK)
    message(FATAL_ERROR "Set WORKBENCH_TRANSCRIPTION_SDK to build the bundled transcription worker")
endif()
find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_path(WHISPER_INCLUDE whisper.h HINTS "${WORKBENCH_TRANSCRIPTION_SDK}/third_party/whisper.cpp/include" REQUIRED)
find_path(GGML_INCLUDE ggml-backend.h HINTS "${WORKBENCH_TRANSCRIPTION_SDK}/third_party/whisper.cpp/ggml/include" REQUIRED)
if(WIN32)
    set(WORKBENCH_ORT_DEFAULT "${WORKBENCH_TRANSCRIPTION_SDK}/third_party/onnxruntime-win-x64-1.25.0")
endif()
set(WORKBENCH_ORT_ROOT "${WORKBENCH_ORT_DEFAULT}" CACHE PATH "ONNX Runtime SDK for the target platform")
find_path(ORT_INCLUDE onnxruntime_cxx_api.h HINTS "${WORKBENCH_ORT_ROOT}/include" REQUIRED)
find_library(WHISPER_LIBRARY whisper HINTS "${WORKBENCH_TRANSCRIPTION_SDK}/build/third_party/whisper.cpp/src" REQUIRED)
find_library(GGML_LIBRARY ggml HINTS "${WORKBENCH_TRANSCRIPTION_SDK}/build/third_party/whisper.cpp/ggml/src" REQUIRED)
find_library(GGML_BASE_LIBRARY ggml-base HINTS "${WORKBENCH_TRANSCRIPTION_SDK}/build/third_party/whisper.cpp/ggml/src" REQUIRED)
find_library(ORT_LIBRARY onnxruntime HINTS "${WORKBENCH_ORT_ROOT}/lib" REQUIRED)
qt_add_executable(TranscriptionWorker modules/transcription/native/worker.cpp
    modules/transcription/native/Controller.cpp modules/transcription/native/Exports.cpp
    modules/transcription/native/Parakeet.cpp modules/transcription/native/resources.qrc)
target_include_directories(TranscriptionWorker PRIVATE ${WHISPER_INCLUDE} ${GGML_INCLUDE} ${ORT_INCLUDE})
target_link_libraries(TranscriptionWorker PRIVATE Qt6::Widgets Qt6::Network ${WHISPER_LIBRARY} ${GGML_LIBRARY} ${GGML_BASE_LIBRARY} ${ORT_LIBRARY})
if(MSVC)
    target_compile_options(TranscriptionWorker PRIVATE /utf-8 /W4)
endif()
endif()

target_link_libraries(ArchaeoHub PRIVATE WorkbenchServices WorkbenchCoordinateDomain Qt6::Quick Qt6::QuickControls2 Qt6::Network)
target_compile_definitions(ArchaeoHub PRIVATE WORKBENCH_VERSION="${PROJECT_VERSION}")
set_target_properties(ArchaeoHub PROPERTIES WIN32_EXECUTABLE TRUE MACOSX_BUNDLE TRUE
    MACOSX_BUNDLE_GUI_IDENTIFIER "org.adocr.archaeology-workbench"
    MACOSX_BUNDLE_BUNDLE_NAME "ArchaeoHub")
if(MSVC)
    foreach(target WorkbenchCore WorkbenchServices ArchaeoHub)
        target_compile_options(${target} PRIVATE /utf-8 /W4)
    endforeach()
endif()
qt_add_translations(ArchaeoHub TS_FILES resources/translations/workbench_en.ts
    RESOURCE_PREFIX /translations)
install(TARGETS ArchaeoHub CoordinateWorker BUNDLE DESTINATION . RUNTIME DESTINATION .)
foreach(module catalog coordinates transcription documents inventory)
    install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules/${module}/engine" DESTINATION modules/${module} OPTIONAL)
endforeach()
install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules/coordinates/spreadsheets" DESTINATION modules/coordinates OPTIONAL)
install(DIRECTORY modules/transcription modules/catalog modules/coordinates modules/documents modules/inventory DESTINATION modules
    FILES_MATCHING PATTERN tool.json)
install(FILES README.md ARCHITECTURE.md MODULES.md BUILDING.md DESIGN.md DESTINATION docs)

endif() # Fin de WORKBENCH_BUILD_APP

if(BUILD_TESTING)
    add_executable(workbench_process_fixture WIN32 tests/process_fixture.cpp)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(workbench_tests tests/core_tests.cpp)
    target_link_libraries(workbench_tests PRIVATE WorkbenchServices Qt6::Test)
    target_compile_definitions(workbench_tests PRIVATE WORKBENCH_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    if(MSVC)
        target_compile_options(workbench_tests PRIVATE /utf-8)
    endif()
    add_test(NAME workbench.contracts COMMAND workbench_tests)
    qt_add_executable(coordinate_text_tests tests/coordinate_text_tests.cpp)
    target_link_libraries(coordinate_text_tests PRIVATE WorkbenchCoordinateDomain Qt6::Test)
    if(MSVC)
        target_compile_options(coordinate_text_tests PRIVATE /utf-8)
    endif()
    add_test(NAME coordinates.text COMMAND coordinate_text_tests)
    if(WORKBENCH_BUILD_APP AND WORKBENCH_BUILD_INTEGRATION_TESTS)
    qt_add_executable(native_controller_tests tests/native_controller_tests.cpp ui/Hub.cpp resources.qrc
        modules/catalog/CatalogController.cpp modules/coordinates/CoordinateController.cpp modules/transcription/TranscriptionController.cpp modules/documents/DocumentController.cpp modules/documents/DocumentLibraryController.cpp modules/inventory/InventoryController.cpp)
    target_link_libraries(native_controller_tests PRIVATE WorkbenchServices WorkbenchCoordinateDomain Qt6::Test Qt6::Network)
    if(MSVC)
        target_compile_options(native_controller_tests PRIVATE /utf-8)
    endif()
    add_test(NAME modules.native_controllers COMMAND native_controller_tests)
    endif()

endif()
